    PROC_FREE=0, PROC_READY, PROC_RUNNING, PROC_WAIT_CHAN,

    uint32_t pid, pc, w, priority;
    t425_proc_state_t state;
} t425_process_t;

typedef struct {
    t425_cpu_t cpu;
    uint8_t ram[T425_RAM_BYTES];
    t425_process_t proc[T425_MAX_PROCESSES];
    uint32_t current_pid;
    uint32_t current_priority;

#include <string.h>
#include "t425_vm.h"

static uint32_t rd32(const uint8_t *p) {
    return (uint32_t)p[0] | ((uint32_t)p[1]<<8) |
           ((uint32_t)p[2]<<16) | ((uint32_t)p[3]<<24);
}
static void wr32(uint8_t *p,uint32_t v) {
    p[0]=(uint8_t)v; p[1]=(uint8_t)(v>>8);
    p[2]=(uint8_t)(v>>16); p[3]=(uint8_t)(v>>24);
}
static void push(t425_cpu_t *c,uint32_t x) {
    c->C=c->B; c->B=c->A; c->A=x;
}
static uint32_t pop(t425_cpu_t *c) {
    uint32_t x=c->A; c->A=c->B; c->B=c->C; return x;
}
static bool in_ram(uint32_t a,uint32_t n) {
    return a>=T425_INT_RAM_BASE && n<=T425_RAM_BYTES &&
           a-T425_INT_RAM_BASE+n<=T425_RAM_BYTES;
}
static uint32_t rw(t425_vm_t *vm,uint32_t a) {
    if(!in_ram(a,4)){vm->cpu.error=1;return 0;}
    return rd32(&vm->ram[a-T425_INT_RAM_BASE]);
}
static void ww(t425_vm_t *vm,uint32_t a,uint32_t v) {
    if(!in_ram(a,4)){vm->cpu.error=1;return;}
    wr32(&vm->ram[a-T425_INT_RAM_BASE],v);
}

/* Prefix semantics from T425 encoding:
   PFIX extends a nibble accumulator; NFIX complements it.
   The following instruction supplies the final nibble. */
static uint32_t operand_value(t425_vm_t *vm,uint8_t nibble) {
    uint32_t v = vm->cpu.prefix_valid ? ((vm->cpu.OPR<<4) | nibble) : nibble;
    vm->cpu.prefix_valid=0;
    vm->cpu.prefix_negative=0;
    vm->cpu.OPR=0;
    return v;
}
static void prefix_pfix(t425_vm_t *vm,uint8_t n) {
    vm->cpu.OPR=(vm->cpu.OPR<<4)|n;
    vm->cpu.prefix_valid=1;
}
static void prefix_nfix(t425_vm_t *vm,uint8_t n) {
    vm->cpu.OPR=~((vm->cpu.OPR<<4)|n);
    vm->cpu.prefix_valid=1;
    vm->cpu.prefix_negative=1;
}

static void operate(t425_vm_t *vm,uint32_t op) {
    t425_cpu_t *c=&vm->cpu; uint32_t x;
    switch(op) {
    case 0x00: x=c->A;c->A=c->B;c->B=x;break;             /* rev */
    case 0x01: {uint32_t a=c->A;if(in_ram(a,1))c->A=vm->ram[a-T425_INT_RAM_BASE];else c->error=1;}break;
    case 0x02: c->A=c->A-c->B; break;                     /* bsub */
    case 0x04: x=pop(c);c->A=x-c->A;break;                /* diff */
    case 0x05: x=pop(c);c->A=x+c->A;break;                /* add */
    case 0x09: x=pop(c);c->A=(x>c->A);break;              /* gt */
    case 0x0A: c->A=(c->A>>2);break;                     /* wsub: word index */
    case 0x0C: x=pop(c);c->A=x-c->A;break;                /* sub */
    case 0x0D: {                                          /* startp: phase-2 model */
        uint32_t pid=pop(c);
        if(pid<T425_MAX_PROCESSES){
            vm->proc[pid].state=PROC_READY;
            vm->proc[pid].pc=vm->proc[pid].pc?vm->proc[pid].pc:c->A;
        }
        break;
    }
    case 0x0F: c->A=~c->A; break;                         /* placeholder safety */
    case 0x12: c->error=1; break;                         /* resetch hook */
    case 0x18: c->A=vm->proc[0].pc;break;
    case 0x19: {                                          /* norm */
        uint32_t v=c->A,n=0;if(v==0){c->A=32;break;}
        while((v&0x80000000u)==0){v<<=1;n++;}
        c->A=n;break;
    }
    case 0x1A: x=pop(c);c->A=(c->A==0)?0:(uint32_t)((int32_t)x/(int32_t)c->A);break;
    case 0x1B: c->A=c->I;break;                           /* ldpi */
    case 0x1D: break;                                     /* xdble hook */
    case 0x1E: c->A=vm->current_priority;break;           /* ldpri */
    case 0x1F: x=pop(c);c->A=(c->A==0)?0:(uint32_t)((int32_t)x%(int32_t)c->A);break;
    case 0x20:                                             /* ret: return address in A */
        c->PC=c->A; c->A=c->B;c->B=c->C;c->C=0; break;
    case 0x21: break;                                     /* lend hook */
    case 0x22: c->A=vm->timer;break;                      /* ldtimer */
    case 0x29: c->error=0; c->A=0;break;                  /* testerr */
    case 0x2A: c->A=vm->analyse?1:0;break;
    case 0x2B: break;                                     /* tin hook */
    case 0x2C: x=pop(c);c->A=(c->A==0)?0:(uint32_t)((int32_t)x/(int32_t)c->A);break;
    case 0x2E: break;                                     /* dist hook */
    case 0x30: break;                                     /* diss */
    case 0x32: c->A=~c->A;break;                          /* not */
    case 0x33: x=pop(c);c->A=x^c->A;break;                /* xor */
    case 0x35: x=pop(c);c->A=x>>c->A;break;               /* lshr approximation */
    case 0x36: x=pop(c);c->A=x<<c->A;break;               /* lshl approximation */
    case 0x38: x=pop(c);c->A=x-c->A;break;                /* lsub */
    case 0x39: vm->proc[vm->current_pid].state=PROC_READY;break; /* runp hook */
    case 0x3A: break;                                     /* xword */
    case 0x3B: {uint32_t a=c->B;if(in_ram(a,1))vm->ram[a-T425_INT_RAM_BASE]=(uint8_t)c->A;else c->error=1;}break;
    case 0x3D: break; case 0x3E: break;
    case 0x40: x=pop(c);c->A=x>>c->A;break;               /* shr */
    case 0x41: x=pop(c);c->A=x<<c->A;break;               /* shl */
    case 0x42: c->A=(int32_t)c->A < 0 ? c->A : 0x7fffffff;break; /* mint model */
    case 0x46: x=pop(c);c->A=x&c->A;break;                /* and */
    case 0x4B: x=pop(c);c->A=x|c->A;break;                /* or */
    case 0x4E: break; case 0x4F: break;
    case 0x50: break; case 0x51: break;
    case 0x52: x=pop(c);c->A=x+c->A;break;                /* sum */
    case 0x53: x=pop(c);c->A=(uint32_t)((int32_t)x*(int32_t)c->A);break;
    case 0x54: vm->timer=0;break;                         /* sttimer */
    case 0x5A: push(c,c->A);break;                        /* dup */
    case 0x63: break; case 0x72: x=pop(c);c->A=x*c->A;break;
    case 0x73: break; case 0x74: break; case 0x75: break;
    case 0x76: {uint32_t v=c->A,n=0;while(v){n+=v&1;v>>=1;}c->A=n;}break;
    case 0x77: {uint32_t v=c->A,r=0;for(int i=0;i<32;i++){r=(r<<1)|(v&1);v>>=1;}c->A=r;}break;
    case 0x78: {uint32_t n=c->A&31,v=pop(c),r=0;for(uint32_t i=0;i<n;i++)r|=((v>>i)&1u)<<(n-1-i);c->A=r;}break;
    case 0x79: pop(c);break;                              /* pop */
    case 0x7A: case 0x7B: case 0x7C: case 0x7D: break;
    case 0x7E: c->A=T425_MEMSTART;break;                 /* ldmemstartval */
    default: c->error=1;c->halted=1;break;
    }
}
void t425_vm_reset(t425_vm_t *vm) {
    memset(vm,0,sizeof(*vm));
    vm->cpu.PC=T425_MEMSTART; vm->cpu.I=T425_MEMSTART; vm->cpu.W=T425_MEMSTART;
    vm->cpu.halted=1;
    for(unsigned i=0;i<T425_MAX_PROCESSES;i++){
        vm->proc[i].pid=i; vm->proc[i].state=(i==0)?PROC_READY:PROC_FREE;
        vm->proc[i].pc=T425_MEMSTART; vm->proc[i].w=T425_MEMSTART;
    }
    vm->current_pid=0; vm->current_priority=0;
}
bool t425_vm_load(t425_vm_t *vm,uint32_t addr,const uint8_t *data,uint32_t len){
    if(!data||!in_ram(addr,len))return false;
    memcpy(&vm->ram[addr-T425_INT_RAM_BASE],data,len);
    vm->cpu.PC=addr;vm->cpu.I=addr;vm->cpu.halted=0;return true;
}
bool t425_vm_read_mem(t425_vm_t *vm,uint32_t addr,uint8_t *out,uint32_t len){
    if(!out||!in_ram(addr,len)) return false;
    memcpy(out,&vm->ram[addr-T425_INT_RAM_BASE],len);
    return true;
}
bool t425_vm_write_mem(t425_vm_t *vm,uint32_t addr,const uint8_t *data,uint32_t len){
    if(!data||!in_ram(addr,len)) return false;
    memcpy(&vm->ram[addr-T425_INT_RAM_BASE],data,len);
    return true;
}
bool t425_vm_step(t425_vm_t *vm){
    t425_cpu_t*c=&vm->cpu;if(c->halted)return false;
    if(!in_ram(c->PC,1)){c->error=1;c->halted=1;return false;}
    uint8_t ins=vm->ram[c->PC++],fn=ins>>4,nib=ins&15;uint32_t op;
    switch(fn){
    case 0x0: op= c->prefix_valid?((c->OPR<<4)|nib):nib; c->prefix_valid=0;c->OPR=0;c->PC+=(int32_t)op;break;
    case 0x1: op=operand_value(vm,nib);push(c,c->W+op*4u);break;
    case 0x2: prefix_pfix(vm,nib);break;
    case 0x3: op=operand_value(vm,nib);c->A=rw(vm,c->A+op*4u);break;
    case 0x4: op=operand_value(vm,nib);push(c,op);break;
    case 0x5: op=operand_value(vm,nib);c->A+=op*4u;break;
    case 0x6: prefix_nfix(vm,nib);break;
    case 0x7: op=operand_value(vm,nib);push(c,rw(vm,c->W+op*4u));break;
    case 0x8: op=operand_value(vm,nib);c->A+=op;break;
    case 0x9: op=operand_value(vm,nib);push(c,c->PC);c->PC+=(int32_t)op;break;
    case 0xA: op=operand_value(vm,nib);if(c->A!=0){c->A=c->B;c->B=c->C;c->PC+=(int32_t)op;}else{c->A=c->B;c->B=c->C;}break;
    case 0xB: op=operand_value(vm,nib);c->W+=(int32_t)op*4;break;
    case 0xC: op=operand_value(vm,nib);c->A=(c->A==op);break;
    case 0xD: op=operand_value(vm,nib);ww(vm,c->W+op*4,pop(c));break;
    case 0xE: op=operand_value(vm,nib);ww(vm,c->A+op*4,c->B);break;
    case 0xF: op=operand_value(vm,nib);operate(vm,op);break;
    }
    vm->instructions++; return !c->halted;
}
void t425_vm_run_slice(t425_vm_t *vm,uint32_t budget){while(budget--&&!vm->cpu.halted)t425_vm_step(vm);}
void t425_vm_get_regs(const t425_vm_t *vm,uint32_t out[6]){
    out[0]=vm->cpu.A;out[1]=vm->cpu.B;out[2]=vm->cpu.C;out[3]=vm->cpu.W;out[4]=vm->cpu.I;out[5]=vm->cpu.PC;
}
void t425_vm_tick(t425_vm_t *vm){ if(vm->timer!=0xFFFFFFFFu) vm->timer++; }
void t425_vm_halt(t425_vm_t *vm){vm->cpu.halted=1;}
void t425_vm_run(t425_vm_t *vm){vm->cpu.halted=0;}

#include "t425_scheduler.h"
#include "t425_vm_scheduler.h"

#include <string.h>
#include <stdio.h>


static bool queue_push(
    uint32_t *queue,
    uint32_t *tail,
    uint32_t *count,
    uint32_t pid
)
{
    if (*count >= T425_MAX_PROCESSES)
        return false;

    queue[*tail] = pid;

    *tail =
        (*tail + 1) %
        T425_MAX_PROCESSES;

    (*count)++;

    return true;
}


static bool queue_pop(
    uint32_t *queue,
    uint32_t *head,
    uint32_t *count,
    uint32_t *pid
)
{
    if (*count == 0)
        return false;

    *pid = queue[*head];

    *head =
        (*head + 1) %
        T425_MAX_PROCESSES;

    (*count)--;

    return true;
}


bool t425_scheduler_init(
    t425_scheduler_t *s
)
{
    if (!s)
        return false;

    memset(s, 0, sizeof(*s));

    s->state =
        T425_SCHED_READY;

    s->current_pid =
        UINT32_MAX;

    return true;
}


bool t425_scheduler_start(
    t425_scheduler_t *s
)
{
    if (!s)
        return false;

    if (s->state !=
        T425_SCHED_READY)
        return false;

    s->state =
        T425_SCHED_RUNNING;

    return true;
}


bool t425_scheduler_enqueue(
    t425_scheduler_t *s,
    uint32_t pid,
    uint32_t priority
)
{
    if (!s)
        return false;

    if (pid >=
        T425_MAX_PROCESSES)
        return false;

    /*
     * Prioridade 0:
     * baixa prioridade.
     *
     * Prioridade != 0:
     * alta prioridade.
     */

    if (priority)
    {
        return queue_push(
            s->high_queue,
            &s->high_tail,
            &s->high_count,
            pid
        );
    }

    return queue_push(
        s->low_queue,
        &s->low_tail,
        &s->low_count,
        pid
    );
}


bool t425_scheduler_dequeue(
    t425_scheduler_t *s,
    uint32_t *pid
)
{
    if (!s || !pid)
        return false;

    /*
     * Processo de alta prioridade
     * sempre vence.
     */

    if (s->high_count)
    {
        return queue_pop(
            s->high_queue,
            &s->high_head,
            &s->high_count,
            pid
        );
    }

    if (s->low_count)
    {
        return queue_pop(
            s->low_queue,
            &s->low_head,
            &s->low_count,
            pid
        );
    }

    return false;
}


bool t425_scheduler_step(

    t425_scheduler_t *s,

    t425_runtime_t *rt

)

{

    /* T425_RUNTIME_TRACE_SCHEDULER */
    static uint32_t t425_trace_sched_calls = 0;

    t425_trace_sched_calls++;

    if ((t425_trace_sched_calls % 100u) == 0u) {

        printf(
            "[TRACE SCHED] "
            "calls=%lu\n",

            (unsigned long)t425_trace_sched_calls
        );
    }

    if (!s ||
        !rt ||
        s->state !=
        T425_SCHED_RUNNING)
        return false;


    /*
     * Clock do runtime.
     */

    t425_runtime_tick(rt);

    s->cycles++;

    s->timer_events++;


    /*
     * Nesta versão o scheduler mantém
     * a compatibilidade com a API antiga.
     *
     * A seleção e execução da VM é feita
     * através da função de integração
     * chamada pelo supervisor.
     *
     * O scheduler continua responsável
     * pelas filas e contabilização.
     */

    return true;
}

bool t425_scheduler_fault(
    t425_scheduler_t *s
)
{
    if (!s)
        return false;

    s->state =
        T425_SCHED_FAULT;

    s->fault_events++;

    return true;
}

#pragma once

#include <stdint.h>

#include "t425_vm.h"

/*
 * Resultado da execução de uma fatia da VM.
 *
 * O scheduler utiliza este resultado para decidir
 * se o processo volta para uma fila READY,
 * permanece bloqueado ou encerra.
 */

typedef enum
{
    T425_VM_SCHED_OK = 0,

    T425_VM_SCHED_YIELD,

    T425_VM_SCHED_BLOCKED,

    T425_VM_SCHED_HALTED,

    T425_VM_SCHED_FAULT

} t425_vm_sched_result_t;


/*
 * Executa um processo específico da VM.
 *
 * pid:
 *     processo arquitetural vm->proc[pid]
 *
 * budget:
 *     quantidade máxima de instruções
 */

t425_vm_sched_result_t
t425_vm_scheduler_execute(

    t425_vm_t *vm,

    uint32_t pid,

    uint32_t budget

);

#include "t425_vm_scheduler.h"

/*
 * A VM atual possui:
 *
 *     vm->current_pid
 *
 * e:
 *
 *     vm->proc[pid]
 *
 * Portanto o scheduler seleciona o PID e a VM
 * executa diretamente aquele processo.
 */


t425_vm_sched_result_t
t425_vm_scheduler_execute(

    t425_vm_t *vm,

    uint32_t pid,

    uint32_t budget

)

{
    if (!vm)
        return T425_VM_SCHED_FAULT;

    if (pid >= T425_MAX_PROCESSES)
        return T425_VM_SCHED_FAULT;


    /*
     * Processo selecionado pelo scheduler.
     */

    vm->current_pid = pid;


    /*
     * Não executar processos livres.
     */

    if (vm->proc[pid].state == PROC_FREE)
        return T425_VM_SCHED_HALTED;


    /*
     * Executa uma fatia limitada.
     */

    if (budget > 0)
        t425_vm_run_slice(

            vm,

            budget

        );


    /*
     * CPU global parada.
     */

    if (vm->cpu.halted)
        return T425_VM_SCHED_HALTED;


    /*
     * Interpretação do estado arquitetural
     * do processo após a execução.
     *
     * READY:
     *     volta ao scheduler.
     *
     * FREE:
     *     processo encerrado.
     *
     * Outros estados:
     *     tratado como bloqueado até o runtime
     *     acordá-lo novamente.
     */

    if (vm->proc[pid].state == PROC_READY)
        return T425_VM_SCHED_YIELD;


    if (vm->proc[pid].state == PROC_FREE)
        return T425_VM_SCHED_HALTED;


    return T425_VM_SCHED_BLOCKED;
}

#ifndef T425_WORKSPACE_WORDS
#define T425_WORKSPACE_WORDS 64

/* Extended process descriptor used by the runtime layer.
   The VM's t425_process_t remains the architectural process record. */
typedef struct {
    uint32_t pid;
    uint32_t pc;
    uint32_t w;
    uint32_t sp;
    uint32_t priority;
    uint32_t workspace[T425_WORKSPACE_WORDS];
    t425_proc_state_t state;
} t425_runtime_process_t;

void t425_process_table_init(t425_runtime_process_t *p);
bool t425_process_start(t425_runtime_process_t *p,uint32_t pid,uint32_t pc,uint32_t priority);
bool t425_process_stop(t425_runtime_process_t *p,uint32_t pid);
bool t425_process_run(t425_runtime_process_t *p,uint32_t pid);
bool t425_process_end(t425_runtime_process_t *p,uint32_t pid);

     */

bool t425_supervisor_start(t425_supervisor_t *s)
{
    if(!s)
        return false;

    /*
     * START idempotente:
     * se já estiver em execução, não é erro.
     */
    if(s->scheduler.state == T425_SCHED_RUNNING)
        return true;

    /*
     * Só podemos iniciar a partir de READY.
     */
    if(s->scheduler.state != T425_SCHED_READY)
        return false;

    if(!t425_scheduler_start(&s->scheduler))
        return false;

    s->runtime.state = T425_RT_RUNNING;

    return true;
}

bool t425_supervisor_step(t425_supervisor_t *s)
{
    if(!s)
        return false;

    /*
     * Não executar o runtime antes do START.
     */
    if(s->scheduler.state != T425_SCHED_RUNNING)
        return false;

    return t425_scheduler_step(
        &s->scheduler,
        &s->runtime
    );
}

bool t425_supervisor_status(
    const t425_supervisor_t *s,
    char *b,
    uint32_t cap
){
    if(!s || !b || cap == 0)
        return false;

    int n = snprintf(
        b,
        cap,
        "NODE=%u RT=%d SCHED=%d TICKS=%lu RX=%lu TX=%lu BAD=%lu",
        s->runtime.config.node_id,
        (int)s->runtime.state,

        (unsigned long)s->runtime.ticks,
        (unsigned long)s->runtime.frames_rx,
        (unsigned long)s->runtime.frames_tx,
        (unsigned long)s->runtime.frames_bad
    );

    return n >= 0 && (uint32_t)n < cap;
}

void t425_occam_runtime_init(t425_occam_runtime_t*r){

    for(int i=0;i<T425_OCCAM_MAX_CHANS;i++){r->chan[i].waiting_in=-1;r->chan[i].waiting_out=-1;}
}
int t425_occam_proc_create(t425_occam_runtime_t*r,uint8_t priority){
    if(!r||r->proc_count>=T425_OCCAM_MAX_PROCS)return -1;
    uint8_t id=r->proc_count++;
    r->proc[id]=(t425_occam_proc_t){.proc_id=id,.state=T425_PROC_READY,.pc=0,.priority=priority};
    return id;
}
bool t425_occam_chan_create(t425_occam_runtime_t*r,uint8_t*id){
    if(!r||!id||r->chan_count>=T425_OCCAM_MAX_CHANS)return false;
    *id=r->chan_count++;
    r->chan[*id].waiting_in=-1;r->chan[*id].waiting_out=-1;
    return true;
}
bool t425_occam_out(t425_occam_runtime_t*r,uint8_t pid,uint8_t cid,const uint8_t*d,uint8_t n){
    if(!r||pid>=r->proc_count||cid>=r->chan_count||n>T425_OCCAM_MAX_VALUE)return false;
    t425_occam_chan_runtime_t*c=&r->chan[cid];
    if(c->count>=T425_OCCAM_QUEUE_DEPTH){
        r->proc[pid].state=T425_PROC_BLOCKED_OUT;c->waiting_out=(int8_t)pid;return false;
    }
    t425_occam_message_t*m=&c->q[c->tail];
    m->chan_id=cid;m->len=n;if(n)memcpy(m->value,d,n);
    c->tail=(uint8_t)((c->tail+1)%T425_OCCAM_QUEUE_DEPTH);c->count++;
    r->proc[pid].state=T425_PROC_READY;
    if(c->waiting_in>=0){r->proc[(uint8_t)c->waiting_in].state=T425_PROC_READY;c->waiting_in=-1;}
    return true;
}
bool t425_occam_in(t425_occam_runtime_t*r,uint8_t pid,uint8_t cid,t425_occam_message_t*m){
    if(!r||!m||pid>=r->proc_count||cid>=r->chan_count)return false;
    t425_occam_chan_runtime_t*c=&r->chan[cid];
    if(c->count==0){r->proc[pid].state=T425_PROC_BLOCKED_IN;c->waiting_in=(int8_t)pid;return false;}
    *m=c->q[c->head];c->head=(uint8_t)((c->head+1)%T425_OCCAM_QUEUE_DEPTH);c->count--;
    r->proc[pid].state=T425_PROC_READY;
    if(c->waiting_out>=0){r->proc[(uint8_t)c->waiting_out].state=T425_PROC_READY;c->waiting_out=-1;}
    return true;
}
void t425_occam_schedule(t425_occam_runtime_t*r){
    if(!r||!r->proc_count)return;
    int best=-1;
    for(uint8_t i=0;i<r->proc_count;i++)
        if(r->proc[i].state==T425_PROC_READY&&(best<0||r->proc[i].priority>r->proc[(uint8_t)best].priority))
            best=i;
    if(best>=0){for(uint8_t i=0;i<r->proc_count;i++)if(r->proc[i].state==T425_PROC_RUNNING)r->proc[i].state=T425_PROC_READY;
        r->proc[(uint8_t)best].state=T425_PROC_RUNNING;r->current=(uint8_t)best;}
}

A ordem acima é o modelo de referência; a implementação atual apenas valida a infraestrutura de ciclo/tick.

----- source/docs/V2_14_SUPERVISOR.md -----
# T425-SIM v2.14 — Supervisor e Scheduler

A v2.14 fecha a coordenação do nó.

```text
                SUPERVISOR
                    │
          ┌─────────┴─────────┐
          ▼                   ▼
      RUNTIME             SCHEDULER
          │                   │
     ┌────┼────┐              │
     ▼    ▼    ▼              ▼
    RX   ACK   TX          TIMER/TICK
     │    │    │              │
     └────┴────┴──────────────┘
                    │
                    ▼
                  STATUS
```

O supervisor:
- inicializa runtime;
- inicializa scheduler;
- inicia o nó;
- executa ciclos;
- expõe status textual;
- permite estado de falha.

O scheduler é deliberadamente simples nesta etapa. A política de prioridade, filas e budget de CPU será definida depois que o comportamento físico do PIO for validado.

## Separação HAL

O supervisor não deve acessar GPIO ou registradores diretamente. A futura camada RP2040 HAL será responsável pelo hardware.

# T425-SIM v1.7 — Runtime OCCAM

A v1.7 implementa o núcleo local de execução:

- processos;
- estados READY/RUNNING/BLOCKED_IN/BLOCKED_OUT/HALTED;
- canais;
- fila de mensagens;
- bloqueio de `IN`;
- bloqueio de `OUT` quando a fila está cheia;
- desbloqueio;
- scheduler por prioridade.

```text
                Scheduler
                    │
       ┌────────────┼────────────┐
       ▼            ▼            ▼
    PROC 0       PROC 1       PROC 2
      │             │
      │ OUT         │ IN
      └──────► CHAN ◄────────────┘
                 │
              queue
```

### Observação semântica

Esta etapa usa uma fila bounded para permitir o desenvolvimento incremental. Ela ainda **não representa o rendezvous estrito OCCAM**, no qual produtor e consumidor sincronizam diretamente. A próxima etapa deve adicionar canais síncronos e canais remotos, preservando o modelo atual como mecanismo de teste.

----- source/docs/V1_8_RENDEZVOUS.md -----
# T425-SIM v1.8 — Rendezvous síncrono

A v1.8 introduz uma máquina de estados para sincronização direta entre `OUT` e `IN`.

```text
EMPTY
 │
 ├── OUT ──► WAIT_OUT ── IN ──► TRANSFER
 │
 └── IN  ──► WAIT_IN  ── OUT ─► TRANSFER
                              │
                              ▼
                            DONE
```

O produtor e o consumidor só completam a operação quando ambos estão presentes.

Esta implementação é o núcleo local do rendezvous. O próximo passo é adicionar identificação do canal e encapsulamento no protocolo remoto, para que o mesmo mecanismo opere entre P00/P01/P02/P03.
